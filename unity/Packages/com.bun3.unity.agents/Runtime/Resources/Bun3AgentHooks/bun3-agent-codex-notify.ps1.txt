# Codex notify -> agent heartbeat, chaining any notify command that was
# installed before us (saved as codex-notify-original.json at install time).
# Codex invokes notify with one JSON argument describing the notification.
param([Parameter(ValueFromRemainingArguments = $true)][string[]]$Rest)

$hooksDir = Join-Path $env:LOCALAPPDATA "bun3-agents\hooks"
$originalFile = Join-Path $hooksDir "codex-notify-original.json"
if (Test-Path $originalFile) {
    try {
        $orig = @(Get-Content $originalFile -Raw | ConvertFrom-Json)
        if ($orig.Count -ge 1) {
            $chainArgs = @()
            if ($orig.Count -gt 1) { $chainArgs += $orig[1..($orig.Count - 1)] }
            if ($Rest) { $chainArgs += $Rest }
            & $orig[0] @chainArgs | Out-Null
        }
    } catch {}
}

# notify fires at turn boundaries: attention-ish -> WaitingInput, else Idle (turn done)
$st = 0
try {
    if ($Rest -and ($Rest[-1] | ConvertFrom-Json).type -match "attention|approval|input") { $st = 3 }
} catch {}

$agents = Join-Path $env:LOCALAPPDATA "bun3-agents\agents"
New-Item -ItemType Directory -Force $agents | Out-Null
$ts = [DateTimeOffset]::UtcNow.ToUnixTimeMilliseconds()
Set-Content -Encoding utf8 (Join-Path $agents "codex-cli.json") `
    "{`"id`":`"codex-cli`",`"n`":`"Codex CLI`",`"st`":$st,`"ts`":$ts,`"pid`":0}"
